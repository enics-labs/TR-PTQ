Project Context:
I am designing a System-on-Chip (SoC) for Machine Learning that uses the MXINT8 (Block Floating Point) format. The architecture consists of a MAC array (dot_product_engine), an MX Requantizer, an MX-to-INT Shifter (mx_dynamic_shifter), a non-linear Vector Processing Unit (tr_nonlinear_vpu), and a Master FSM Controller (tr_soc_ctrl_mx).

Architectural Rules Established:

The VPU Datapath: The VPU SRAM interfaces are strictly 8-bit, but the internal crossbar and math backbone (Softmax/GELU) operate in 16-bit space (W_MAC=16) to prevent integer clamping and preserve true magnitude.

Top-Level Expansion: The mx_dynamic_shifter sits at the Top Level. It dynamically shifts 8-bit mantissas into 16-bit integers using the MX shared exponent before the data enters the VPU.

FSM Control: The controller issues a ctrl_enable_linear_shift flag to activate the Top-Level shifter only during scale-variant non-linear passes (GELU, Softmax). For scale-invariant passes (RMSNorm) or standard MAC writes, the shifter is bypassed.

Shared ROM: The VPU uses a shared_lut_rom that is 8-bit. We use SystemVerilog $signed casting to dynamically pad it to 16-bit when entering the VPU crossbar.

module tr_nonlinear_vpu #(
    parameter int N         = 8,
    parameter int W_VEC     = 16,  // Set to 16 for MX mode via Top Level
    parameter int W_MAC     = 16,
    parameter int ACC_W     = 32,
    parameter int FRAC_W    = 4,
    parameter int LUT_IDX_W = 3
)(
    input  logic clk, rst_n,

    input  logic signed [W_VEC-1:0] sram_data_a [N],
    input  logic signed [W_VEC-1:0] sram_data_b [N],
    output logic signed [W_VEC-1:0] vpu_data_out [N],
    
    input  logic signed [W_VEC-1:0] ctrl_scalar_sub_val,
    output logic signed [W_VEC-1:0] vpu_max_out,         
    output logic signed [ACC_W-1:0] vpu_dot_out,         

    input  logic [2:0]  mux_bb_in_sel,
    input  logic [1:0]  mux_mac_a_sel, mux_mac_b_sel,
    input  logic [1:0]  mux_vecmul_a_sel, mux_vecmul_b_sel,
    input  logic [2:0]  mux_vpu_out_sel,

    input  logic        en_piped_max, en_mac_valid, en_vecmul_valid, en_bb_valid,
    input  logic        mac_clear_acc, bb_shift_mode, bb_bypass_ln, bb_mode_pre_ln, sym_mode_en,
    input  logic [1:0]  mac_op_mode, vecmul_op_mode, vecmul_scale_mode, bb_mode_post_ln,

    output logic        vpu_bb_valid_out, vpu_vecmul_valid_out, vpu_mac_valid_out
);

    logic signed [W_VEC-1:0] scalar_sub_out [N], alpha_stab_out [N], sym_mod_out [N];
    logic signed [W_MAC-1:0] bb_in [N];
    logic signed [W_VEC-1:0] bb_log [N];
    logic        [LUT_IDX_W-1:0] bb_a_idx [N];
    logic        [W_VEC-1:0] bb_mantisa [N];
    logic                    bb_is_zero [N];
    logic        [7:0]       rom_e_a [N]; 
    
    logic        [W_VEC-1:0] mac_in_a [N], mac_in_b [N];
    logic        [W_VEC-1:0] vecmul_in_a [N], vecmul_in_b [N];
    logic signed [ACC_W-1:0] vecmul_out_acc [N];
    logic signed [W_VEC-1:0] vecmul_out_trunc [N];
    logic int_bb_valid, int_mac_valid, int_vecmul_valid;

    // Safe dynamic padding for the 8-bit ROM
    logic signed [W_VEC-1:0] rom_e_a_padded [N];
    always_comb begin
        for (int i=0; i<N; i++) begin
            if (bb_is_zero[i]) 
                rom_e_a_padded[i] = W_VEC'(signed'(-8'sd128));
            else 
                rom_e_a_padded[i] = W_VEC'(signed'({1'b0, rom_e_a[i]}));
        end
    end

    always_comb begin
        for (int i = 0; i < N; i++) begin
            case (mux_bb_in_sel)
                3'b000: bb_in[i] = bb_shift_mode ? vpu_dot_out[19:4] : vpu_dot_out[23:8];
                3'b001: bb_in[i] = W_MAC'(alpha_stab_out[i]);
                3'b010: bb_in[i] = W_MAC'(scalar_sub_out[i]);
                3'b011: bb_in[i] = W_MAC'(sram_data_a[i]);
                3'b100: bb_in[i] = W_MAC'(ctrl_scalar_sub_val);
                default: bb_in[i] = '0;
            endcase

            case (mux_mac_a_sel)
                2'b00: mac_in_a[i] = sram_data_a[i];
                2'b01: mac_in_a[i] = rom_e_a_padded[i];
                2'b10: mac_in_a[i] = scalar_sub_out[i];
                default: mac_in_a[i] = sram_data_a[i];
            endcase

            case (mux_mac_b_sel)
                2'b00: mac_in_b[i] = sram_data_b[i];
                2'b01: mac_in_b[i] = bb_is_zero[i] ? W_VEC'(signed'(bb_mantisa[i]) << 1) : W_VEC'(signed'(bb_mantisa[i])); 
                2'b10: mac_in_b[i] = sram_data_a[i];
                default: mac_in_b[i] = sram_data_b[i];
            endcase

            case (mux_vecmul_a_sel)
                2'b00: vecmul_in_a[i] = sram_data_a[i];
                2'b01: vecmul_in_a[i] = scalar_sub_out[i];
                2'b10: vecmul_in_a[i] = W_VEC'(signed'(bb_mantisa[0]));
                2'b11: vecmul_in_a[i] = rom_e_a_padded[i]; 
            endcase

            case (mux_vecmul_b_sel)
                2'b00: vecmul_in_b[i] = sram_data_b[i];
                2'b01: vecmul_in_b[i] = sym_mod_out[i];
                2'b10: vecmul_in_b[i] = bb_is_zero[i] ? W_VEC'(signed'(bb_mantisa[i]) << 1) : W_VEC'(signed'(bb_mantisa[i]));
                default: vecmul_in_b[i] = sram_data_b[i];    
            endcase
        end
    end

    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) for(int i=0; i<N; i++) vpu_data_out[i] <= '0;
        else begin
            for(int i=0; i<N; i++) begin
                case (mux_vpu_out_sel)
                    3'b000: vpu_data_out[i] <= vecmul_out_trunc[i];
                    3'b001: vpu_data_out[i] <= bb_log[i]; 
                    3'b010: vpu_data_out[i] <= alpha_stab_out[i];
                    3'b011: vpu_data_out[i] <= scalar_sub_out[i];
                    default: vpu_data_out[i] <= vecmul_out_trunc[i];
                endcase
            end
        end
    end

    piped_max #(.NUM_INPUTS(N), .DATA_WIDTH(W_VEC)) u_max (.clk(clk), .rst_n(rst_n), .valid_in(en_piped_max), .in_data(sram_data_a), .max_out(vpu_max_out), .valid_out());
    scalar_sub #(.NUM_INPUTS(N), .DATA_WIDTH(W_VEC)) u_sub (.in_data(sram_data_a), .sub_val(ctrl_scalar_sub_val), .out_data(scalar_sub_out));
    alpha_stabilizer #(.N(N), .W(W_VEC)) u_alpha (.in_vec(sram_data_a), .out_vec(alpha_stab_out));
    symmetry_modifier #(.N(N), .WIDTH_X(W_VEC), .WIDTH_Y(W_VEC), .FRAC_W(FRAC_W)) u_sym (.x_raw(sram_data_a), .y_sig(sram_data_b), .mode_en(sym_mode_en), .sig_corrected(sym_mod_out));
    tr_backbone_wrapper #(.N(N), .WIDTH_IN(W_MAC), .WIDTH_OUT(W_VEC), .FRAC_W(FRAC_W), .LUT_IDX_W(LUT_IDX_W)) u_backbone (.clk(clk), .rst_n(rst_n), .in_valid(en_bb_valid), .out_valid(int_bb_valid), .bypass_ln(bb_bypass_ln), .mode_pre_ln(bb_mode_pre_ln), .mode_post_ln(bb_mode_post_ln), .vec_in(bb_in), .log_out(bb_log), .a_idx_out(bb_a_idx), .mantisa_out(bb_mantisa), .is_zero_out(bb_is_zero));
    shared_lut_rom #(.N(N)) u_rom (.a_idx(bb_a_idx), .e_a(rom_e_a));
    mac_array_engine #(.N(N), .W(W_VEC), .ACC_W(ACC_W)) u_mac (.clk(clk), .rst_n(rst_n), .in_valid(en_mac_valid), .in_ready(), .op_mode(mac_op_mode), .a(mac_in_a), .b(mac_in_b), .c('0), .clear_acc(mac_clear_acc), .out_valid(int_mac_valid), .out_ready(1'b1), .out_dot(vpu_dot_out));
    vec_mul_array_engine #(.N(N), .W(W_VEC), .ACC_W(ACC_W)) u_vecmul (.clk(clk), .rst_n(rst_n), .in_valid(en_vecmul_valid), .in_ready(), .op_mode(vecmul_op_mode), .a(vecmul_in_a), .b(vecmul_in_b), .out_valid(int_vecmul_valid), .out_ready(1'b1), .out_vec(vecmul_out_acc));

    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin vpu_bb_valid_out <= 0; vpu_mac_valid_out <= 0; vpu_vecmul_valid_out <= 0; end 
        else begin vpu_bb_valid_out <= int_bb_valid; vpu_mac_valid_out <= int_mac_valid; vpu_vecmul_valid_out <= int_vecmul_valid; end
    end

    always_comb begin
        for(int i=0; i<N; i++) begin
            if (vecmul_scale_mode == 2'b10) vecmul_out_trunc[i] = vecmul_out_acc[i][15:8];
            else if (vecmul_scale_mode == 2'b01) vecmul_out_trunc[i] = vecmul_out_acc[i][11:4];
            else vecmul_out_trunc[i] = vecmul_out_acc[i][7:0];
        end
    end
endmodule

module tr_soc_top_mx #(
    parameter int M        = 4,
    parameter int N        = 8,
    parameter int W        = 8,
    parameter int ACC_W    = 32,
    parameter int VPU_IN_W = 16  // 16-bit expansion
)(
    // ... standard generic IO ...
);

    // INTERNAL ROUTING
    logic signed [W-1:0] req_vec_out [M];
    logic signed [7:0] mx_shared_exp;
    logic signed [VPU_IN_W-1:0] shifted_linear_data [M]; 
    logic signed [VPU_IN_W-1:0] vpu_sram_a_in [N];
    logic signed [VPU_IN_W-1:0] vpu_sram_b_in [N];
    logic ctrl_enable_linear_shift;
    // ... FSM control wires ...

    // The FSM Controller
    tr_soc_ctrl_mx #(.N(N), .W(W), .VPU_W(VPU_IN_W), .ACC_W(ACC_W)) u_ctrl (.*);
    
    // The MX Shifter (Expands M lanes from 8-bit to 16-bit based on exponent)
    mx_dynamic_shifter #(
        .N(M), .IN_W(W), .OUT_W(VPU_IN_W)
    ) u_shifter_expand (
        .data_in(req_vec_out),
        .shift_amount(mx_shared_exp),
        .shift_dir(1'b1),
        .data_out(shifted_linear_data)
    );

    // VPU Input Routing (Handles the 16-bit expanded data)
    always_comb begin
        for(int i = 0; i < N; i++) begin
            if (i < M) begin
                automatic logic signed [VPU_IN_W-1:0] sram_a_route;
                sram_a_route = ctrl_enable_linear_shift ? shifted_linear_data[i] : VPU_IN_W'(signed'(req_vec_out[i]));
                vpu_sram_a_in[i] = (src_sram_a_sel) ? ctrl_scratch_a[i] : sram_a_route;
            end else begin
                vpu_sram_a_in[i] = '0;
            end
            vpu_sram_b_in[i] = (src_sram_b_sel) ? ctrl_scratch_b[i] : VPU_IN_W'(signed'(ext_sram_b[i]));
        end
    end

    // The VPU (Operating at W_VEC = VPU_IN_W = 16)
    tr_nonlinear_vpu #(
        .N(N), .W_VEC(VPU_IN_W), .W_MAC(16), .ACC_W(ACC_W), .FRAC_W(4), .LUT_IDX_W(3)
    ) u_vpu (
        .sram_data_a(vpu_sram_a_in), .sram_data_b(vpu_sram_b_in),
        // ... rest of ports
    );
endmodule
