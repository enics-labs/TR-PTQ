\subsection{Segmented Online Normalizer for Stable Softmax}

Let $X = (x_1, x_2, \dots, x_N)$ be a vector. Softmax is defined as
\begin{equation}
\mathrm{softmax}(x_i) = \frac{e^{x_i - x_{\max}}}{\sum_{j=1}^{N} e^{x_j - x_{\max}}},
\qquad
x_{\max} = \max_{1 \le j \le N} x_j .
\end{equation}
Define the stabilized normalizer
\begin{equation}
S \triangleq \sum_{j=1}^{N} e^{x_j - x_{\max}}.
\end{equation}
A naïve implementation requires multiple passes over $X$ (to find $x_{\max}$, to compute $S$, and optionally to normalize the outputs). We instead formalize a segmented online computation that supports higher parallelism.

\paragraph{Segmentation.}
Partition $X$ into $m$ (contiguous) segments
\begin{equation}
X = \bigcup_{k=1}^{m} X^{(k)}, 
\qquad 
X^{(k)} = \{x_{k,1}, x_{k,2}, \dots, x_{k,n_k}\},
\qquad
\sum_{k=1}^{m} n_k = N .
\end{equation}

\paragraph{Local maxima and running maximum.}
For each segment $k$, define the local maximum
\begin{equation}
\ell_k \triangleq \max_{x \in X^{(k)}} x,
\end{equation}
and the running (global-so-far) maximum
\begin{equation}
M_k \triangleq \max(M_{k-1}, \ell_k),
\qquad
M_0 \triangleq -\infty .
\end{equation}
Hence, $M_k$ equals the maximum over the prefix $X^{(1)} \cup \cdots \cup X^{(k)}$.

\paragraph{Local exponential sum.}
Within each segment, compute a locally stabilized sum
\begin{equation}
S_k \triangleq \sum_{x \in X^{(k)}} e^{x - \ell_k}.
\end{equation}
This computation is fully parallelizable within the segment.

\paragraph{Online update of the global normalizer.}
Let
\begin{equation}
T_k \triangleq \sum_{i=1}^{k} \sum_{x \in X^{(i)}} e^{x - M_k}
\end{equation}
denote the stabilized normalizer for the prefix up to segment $k$, referenced to the current running maximum $M_k$.
Using $e^{x - M_k} = e^{x - \ell_k}\, e^{\ell_k - M_k}$ and the change of reference from $M_{k-1}$ to $M_k$, we obtain the recurrence
\begin{equation}
\boxed{
T_k
=
T_{k-1}\, e^{M_{k-1} - M_k}
+
S_k\, e^{\ell_k - M_k}
},
\qquad
T_0 \triangleq 0 .
\end{equation}

\paragraph{Final result.}
After processing all $m$ segments,
\begin{equation}
M_m = x_{\max},
\qquad
T_m = \sum_{j=1}^{N} e^{x_j - x_{\max}} = S,
\end{equation}
i.e., $T_m$ equals the exact softmax denominator computed in a single streaming pass with segment-level parallelism.
